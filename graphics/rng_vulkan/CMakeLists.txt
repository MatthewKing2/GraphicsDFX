cmake_minimum_required(VERSION 3.20)
project(GraphicsDFXRng LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Vulkan REQUIRED)
find_package(glfw3 3.3 CONFIG QUIET)
if(NOT TARGET glfw)
    if(WIN32)
        set(GLFW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../Libraries/glfw-3.4.bin.WIN64")
        find_library(GLFW_LIBRARY NAMES glfw3 HINTS "${GLFW_ROOT}/lib-vc2022" REQUIRED)
        add_library(glfw UNKNOWN IMPORTED)
        set_target_properties(glfw PROPERTIES IMPORTED_LOCATION "${GLFW_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${GLFW_ROOT}/include"
            INTERFACE_LINK_LIBRARIES "user32;gdi32;shell32")
    else()
        message(FATAL_ERROR "Install GLFW 3.3 or newer")
    endif()
endif()
set(SHADER_DIR "${CMAKE_CURRENT_BINARY_DIR}/shaders")
find_program(RNG_SHADER_COMPILER NAMES glslc glslangValidator HINTS "$ENV{VULKAN_SDK}/bin")
if(NOT RNG_SHADER_COMPILER)
    message(FATAL_ERROR "Install glslc (shaderc) or glslangValidator (glslang-tools)")
endif()
get_filename_component(SHADER_COMPILER_NAME "${RNG_SHADER_COMPILER}" NAME)
if(SHADER_COMPILER_NAME MATCHES "^glslangValidator")
    set(SHADER_COMPILER_ARGS -V --target-env vulkan1.1 -Os "-I${CMAKE_CURRENT_SOURCE_DIR}/shaders")
else()
    set(SHADER_COMPILER_ARGS -O --target-env=vulkan1.1 -I "${CMAKE_CURRENT_SOURCE_DIR}/shaders")
endif()
message(STATUS "RNG shader compiler: ${RNG_SHADER_COMPILER}")
foreach(SHADER rng.comp simulate.comp visual.comp display.vert display.frag)
    add_custom_command(OUTPUT "${SHADER_DIR}/${SHADER}.spv"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${SHADER_DIR}"
        COMMAND "${RNG_SHADER_COMPILER}" ${SHADER_COMPILER_ARGS}
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}" -o "${SHADER_DIR}/${SHADER}.spv"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${SHADER}"
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders/common.glsl"
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders/philox.glsl" VERBATIM)
    list(APPEND SHADER_BINARIES "${SHADER_DIR}/${SHADER}.spv")
endforeach()
add_custom_target(rng_shaders DEPENDS ${SHADER_BINARIES})
add_library(random_source src/random_source.cpp)
target_include_directories(random_source PUBLIC src)
add_executable(rng_vulkan src/main.cpp src/gpu.cpp)
target_link_libraries(rng_vulkan PRIVATE Vulkan::Vulkan glfw random_source)
target_compile_definitions(rng_vulkan PRIVATE RNG_SHADER_DIR="${SHADER_DIR}")
add_dependencies(rng_vulkan rng_shaders)
if(MSVC)
    target_compile_options(rng_vulkan PRIVATE /W4 /fp:strict)
else()
    target_compile_options(rng_vulkan PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers -ffp-contract=off)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(rng_reference_tests tests/reference_test.cpp)
    target_include_directories(rng_reference_tests PRIVATE src)
    if(MSVC)
        target_compile_options(rng_reference_tests PRIVATE /fp:strict)
    else()
        target_compile_options(rng_reference_tests PRIVATE -ffp-contract=off)
    endif()
    add_test(NAME rng_reference COMMAND rng_reference_tests)
    add_executable(random_source_tests tests/random_source_test.cpp)
    target_link_libraries(random_source_tests PRIVATE random_source)
    add_test(NAME random_stream_protocol COMMAND random_source_tests)
    add_test(NAME invalid_arguments COMMAND rng_vulkan --count 0)
    set_tests_properties(invalid_arguments PROPERTIES WILL_FAIL TRUE)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME profiling_reports COMMAND ${Python3_EXECUTABLE} -m unittest discover
            -s "${CMAKE_CURRENT_SOURCE_DIR}/tools" -p "test_*.py")
        add_test(NAME random_stream_generator COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/tools/make_rng_stream.py" --self-test)
    endif()
endif()
